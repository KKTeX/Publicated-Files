%  \名前マーク配列[n] という TeX マクロを生やす。n 番目だけ黒く塗られる。

%  \マークシート は横一列にしか並べられないので、縦に積む受験番号欄は
%  「1 要素の配列を 1 セルずつ」置いて列を作る。これで \jegg の描画は
%  すべて正式機能（\マークシート）経由のままにできる。

\MarkArrayMake{単0}{|0|}\MarkArrayMake{単1}{|1|}\MarkArrayMake{単2}{|2|}

% ---- 2.4 マーク例 ---------------------------------------------------
%  良い例は \マークシート の第3引数（正解位置）で 1 番目を黒く塗る。
%  悪い例は「楕円の中に何が書いてあるか」で見せるので、配列の要素に

\MarkArrayMake{良例}{| |}
\MarkArrayMake{悪例}{|・|レ|$\times$|}

% ---- 3.0 KKran ブロックの配置 ---------------------------------------
%  \KKran が返す箱は、描画領域の周りに左右対称の余白を持つ
%  （\KKran[0.9pt] のとき片側およそ 1.4mm）。anchor=north west で置くと
%  その余白ぶん絵が右へずれるので、「中心」を指定して置く。

%  \CTplace{左端}{上端}{幅}{高さ}{ブロック}
\newcommand{\CTplace}[5]{%
  \node[anchor=center, inner sep=0pt] at ({#1+0.5*#3},{#2-0.5*#4}) {#5};%
}
